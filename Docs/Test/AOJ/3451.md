点の部分集合を選んで凸包を構築し、外周及び内部に含まれる点の重みの和を最大化する問題。

colinearでない凸包dpをコンテスト中に1から実装するのは無理な気がするので、このコードを改変すると良い。

改変しそうなところをメモしておく。


```cpp
int N;
cin >> N;
PointCloud P(N);
vector<long long> W(N);
for (int i = 0 ; i < N ; i++)
    cin >> P[i] >> W[i];
TriangleProduct<AdditiveGroup<long long>> cpit{P,W}; // <- ここの重みの掛け合わせかた
const long long INF = (long long)-1e18;
auto makeDP = [&](vector<pair<int,int>> ord) -> vector<vector<long long>> {
    vector dp(N,vector<long long>(N,INF));
    for (int i = 0 ; i < N ; i++)
        dp[i][i] = cpit.exact(i).first;
    for (auto [mid,to] : ord) {
        assert(P[mid] != P[to]);
        // 別の点をまたいでいるベクトルはスキップする
        if (cpit.onLine(mid,to).second)
            continue;
        for (int from = 0 ; from < N ; from++)
            if (dp[from][mid] != INF) {
                // cpit.product(from,mid,to)は変わる可能性が高そう
                // Integer Convex Hullだと三角形の面積だし、最古の遺跡だと問答無用で0
                long long add = cpit.product(from,mid,to).first+cpit.onLine(mid,to).first+cpit.exact(to).first;
                if (Cross(P[mid]-P[from],P[to]-P[mid]))
                    add += cpit.onLine(from,to).first;
                dp[from][to] = max(dp[from][to],dp[from][mid]+add);
            }
    }
    return dp;
};
// dn,upの構築は基本的に同じで良いハズ。
auto dp = makeDP(dn), ep = makeDP(up);
long long ans = *ranges::max_element(W); // <- 1点のみはここで別処理している
for (int i = 0 ; i < N ; i++)
    for (int j = 0 ; j < N ; j++)
        if (dp[i][j] != INF)
            ans = max(ans,dp[i][j]+ep[j][i]-cpit.exact(i).first-cpit.exact(j).first-cpit.onLine(i,j).first);
            // 線分(i,j)上の点は丁度二回寄与しているので、これを外す
cout << ans << '\n';
```
